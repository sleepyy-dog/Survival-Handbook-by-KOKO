> #### **题目 (第$6$次作业 第$1$题):**
>
> 分别用 $fzero$ 和 $fsolve$ 程序求方程 $\sin x - x^2/2 = 0$ 的所有根，准确到 $10^{-10}$，取不同的初值计算，输出初值、根的近似值和迭代次数，分析不同根的收敛域；自己构造某个迭代公式（如 $x=(2\sin x)^{1/2}$ 等）用迭代法求解，并自己编写牛顿法的程序进行求解和比较。

**解：**

结果：

- $fzero$：
  - 对多组初值收敛至 $x=0$ 或 $x^\star\approx 1.4044148241$，迭代步数约为 $5\sim 7$ 步
- $fsolve$：
  - 对多组初值均收敛至$x=0$ 或$x^\star\approx 1.4044148241$，迭代步数约为 $5\sim 7$ 步
- 固定点迭代：
  - $g_0(x)$ 在 $x=0$ 附近收敛快（$g_0'(0)=0$），$|x_0|\le 0.5$ 的测试均在 $4\sim 6$ 步达标；
  - $g_+(x)=\sqrt{2\sin x}$ 对正根线性收敛，迭代步数约 $11\sim 14$ 步；若 $x_0=0$，会立即停在 $0$。
- 自编牛顿法：
  - 典型迭代步数 $3\sim 5$ 步即可达 $10^{-10}$，对两枚根均二次收敛，表现与 $fzero$ 相当且更快于固定点迭代。

* 收敛域分析

  - 牛顿法：$401$ 个初值中，收敛到 $x=0$ 的有 $274$ 个，收敛到 $x^\star$ 的有 $121$ 个，有 $6$ 个初值失败。


  - $fzero$：到 $x=0$ 的 $274$ 个，到 $x^\star$ 的 $122$ 个，有 $5$ 个失败点。


自构函数：

$$
x=\sqrt{2\sin x}
$$

%% 定义问题
F  = @(x) sin(x) - 0.5*x.^2;
dF = @(x) cos(x) - x;

% 全局精度与迭代上限
ABS_TOL = 1e-10;
REL_TOL = Inf;
MAX_ITER = 200;

% 参考根（用于吸引域分类）
opt_fzero = optimset('TolX', ABS_TOL, 'Display', 'off');
x_root0 = 0.0;
x_rootp = fzero(F, [1, 2], opt_fzero);

%% 一、fzero 求解（多初值）
x0_list_fzero = [-1.0, -0.5, -0.1, 0.0, 0.1, 0.5, 1.0, 1.3, 1.5, 2.0];
x_fzero = zeros(size(x0_list_fzero));
k_fzero = zeros(size(x0_list_fzero));

for k = 1:length(x0_list_fzero)
    x0 = x0_list_fzero(k);
    [x_sol, ~, ~, output] = fzero(F, x0, opt_fzero);
    x_fzero(k) = x_sol;
    k_fzero(k) = output.iterations;
end

fprintf('fzero 求解 sin(x)-x^2/2=0（TolX=1e-10）：\n');
for k = 1:length(x0_list_fzero)
    fprintf('x0(%d) = % .6f, 根 = % .10f, 迭代次数：%d\n', k, x0_list_fzero(k), x_fzero(k), k_fzero(k));
end
fprintf('\n');

%% 二、fsolve 求解（多初值，标量）
options_fsolve = optimoptions('fsolve', 'Display', 'off', ...
    'FunctionTolerance', ABS_TOL, 'StepTolerance', REL_TOL);

x0_list_fsolve = x0_list_fzero;
x_fsolve = zeros(size(x0_list_fsolve));
k_fsolve = zeros(size(x0_list_fsolve));

for k = 1:length(x0_list_fsolve)
    x0 = x0_list_fsolve(k);
    [x_sol, ~, ~, output] = fsolve(F, x0, options_fsolve);
    x_fsolve(k) = x_sol;
    k_fsolve(k) = output.iterations;
end

fprintf('fsolve 求解 sin(x)-x^2/2=0（FunctionTol=1e-10, StepTol=1e-10）：\n');
for k = 1:length(x0_list_fsolve)
    fprintf('x0(%d) = % .6f, 根 = % .10f, 迭代次数：%d\n', k, x0_list_fsolve(k), x_fsolve(k), k_fsolve(k));
end
fprintf('\n');

%% 三、固定点迭代
% 1) 收敛到零根：g0(x) = x - (sin x - x^2/2)
g0 = @(x) x - (sin(x) - 0.5*x.^2);
x0_list_fp0 = [-0.5, -0.2, -0.05, 0.0, 0.05, 0.2, 0.5];
x_fp0 = zeros(size(x0_list_fp0));
k_fp0 = zeros(size(x0_list_fp0));
for k = 1:length(x0_list_fp0)
    [x_sol, kk] = fixed_point(g0, F, x0_list_fp0(k), MAX_ITER, ABS_TOL);
    x_fp0(k) = x_sol; k_fp0(k) = kk;
end
fprintf('固定点迭代 g0(x)=x-(sin x - x^2/2)（目标：零根）：\n');
for k = 1:length(x0_list_fp0)
    fprintf('x0(%d) = % .6f, 根 = % .10f, 迭代次数：%d\n', k, x0_list_fp0(k), x_fp0(k), k_fp0(k));
end
fprintf('\n');

% 2) 收敛到正根：g+(x) = sqrt(2 sin x)（只在 sin x >= 0 有效）
gpos = @(x) sqrt(max(2*sin(x), 0));
x0_list_fpp = [0.0, 0.1, 0.3, 0.7, 1.0, 1.3];
x_fpp = zeros(size(x0_list_fpp));
k_fpp = zeros(size(x0_list_fpp));
for k = 1:length(x0_list_fpp)
    [x_sol, kk] = fixed_point(gpos, F, x0_list_fpp(k), MAX_ITER, ABS_TOL);
    x_fpp(k) = x_sol; k_fpp(k) = kk;
end
fprintf('固定点迭代 g+(x)=sqrt(2 sin x)（目标：正根）：\n');
for k = 1:length(x0_list_fpp)
    fprintf('x0(%d) = % .6f, 根 = % .10f, 迭代次数：%d\n', k, x0_list_fpp(k), x_fpp(k), k_fpp(k));
end
fprintf('\n');

%% 四、自编牛顿法求解并比较
x0_list_newton = [-1.0, -0.5, -0.1, 0.0, 0.1, 0.5, 1.0, 1.3, 1.5, 2.0];
x_newton = zeros(size(x0_list_newton));
k_newton = zeros(size(x0_list_newton));

for k = 1:length(x0_list_newton)
    [x_sol, kk] = newton_scalar(F, dF, x0_list_newton(k), MAX_ITER, ABS_TOL, REL_TOL);
    x_newton(k) = x_sol; k_newton(k) = kk;
end

fprintf('自编牛顿法（abs_tol=1e-10, rel_tol=1e-10）：\n');
for k = 1:length(x0_list_newton)
    fprintf('x0(%d) = % .6f, 根 = % .10f, 迭代次数：%d\n', k, x0_list_newton(k), x_newton(k), k_newton(k));
end

%% 五、收敛域分析（吸引域）：在区间 [-2, 2] 扫描初值并分类（Newton 与 fzero）
x0_scan = linspace(-2, 2, 401);

% Newton 扫描
idN = nan(size(x0_scan));   % 1: 零根, 2: 正根
kN  = nan(size(x0_scan));
for i = 1:length(x0_scan)
    try
        [xN, kk] = newton_scalar(F, dF, x0_scan(i), MAX_ITER, ABS_TOL, REL_TOL);
        kN(i) = kk;
        if abs(xN - x_root0) <= 1e-10
            idN(i) = 1;
        elseif abs(xN - x_rootp) <= 1e-10
            idN(i) = 2;
        end
    catch
        % 保持 NaN
    end
end

% fzero 扫描
idZ = nan(size(x0_scan));
kZ  = nan(size(x0_scan));

    try
        [xZ, ~, ~, outZ] = fzero(F, x0_scan(i), opt_fzero);
        kZ(i) = outZ.iterations;
        if abs(xZ - x_root0) <= 1e-10
            idZ(i) = 1;
        elseif abs(xZ - x_rootp) <= 1e-10
            idZ(i) = 2;
        end
    catch
        % 保持 NaN
    end
end

% 统计输出
fprintf('收敛域（Newton）：到零根的初值数 = %d，到正根的初值数 = %d\n', sum(idN==1), sum(idN==2));
fprintf('收敛域（fzero ）：到零根的初值数 = %d，到正根的初值数 = %d\n', sum(idZ==1), sum(idZ==2));
%% 局部函数
function [x, k] = newton_scalar(F, dF, x0, max_iter, abs_tol, rel_tol)
    x = x0;
    k = 0;
    rel_err = rel_tol + 1;
    Fx = F(x);
    while (abs(Fx) > abs_tol || rel_err > rel_tol) && k < max_iter
        Jx = dF(x);
        if abs(Jx) < 1e-14
            error('导数过小，可能导致数值不稳定');
        end
        x_new = x - Fx / Jx;
        rel_err = abs(x_new - x) / max(1, abs(x_new));
        x = x_new;
        Fx = F(x);
        k = k + 1;
    end
    if k >= max_iter
        error('未在最大迭代次数内收敛');
    end
end

function [x, k] = fixed_point(g, F, x0, max_iter, tol)
    x = x0;
    for k = 1:max_iter
        x_new = g(x);
        if ~isfinite(x_new)
            break
        end
        if abs(x_new - x) < tol && abs(F(x_new)) < tol
            x = x_new; return
        end
        x = x_new;
    end
end
